# Optional extras. Nothing here is needed to run Wizard.
#
# Install with:  uv pip install --system -r requirements-optional.txt
-i https://pkgs.safetycli.com/repository/nil-2662d/project/ai_data_analyst/pypi/simple/

# Shared cache and job state across workers. Activated only when REDIS_URL is
# set; without it the app uses an in-process cache and queue.
redis>=5.0.0

# Talk to an OpenAI-compatible endpoint instead of a local Ollama daemon
# (API_PROVIDER=openai, custom_gateway, or lmstudio).
langchain-openai>=0.1.0

# Anthropic's Messages API, which is not OpenAI-shaped and needs its own client.
# Only required when a role is assigned to the `anthropic` provider; without it
# that provider reports what to install rather than failing obscurely.
langchain-anthropic>=0.1.0

# ---------------------------------------------------------------------------
# Data source connectors
# ---------------------------------------------------------------------------
# Drivers for connecting to a database instead of uploading a file. Optional
# because most installs never connect to anything, and none of these belongs in
# the API image for someone who only analyses CSVs.
#
# A missing driver is reported, never fatal: the connection kind is still listed
# with the pip command that enables it, which is also how the UI decides whether
# to offer a button or an instruction.

# Every relational engine at once -- Postgres, MySQL, SQL Server, Snowflake and
# the rest are SQLAlchemy *dialects*, so which databases an install can reach is
# a question of which dialect driver is also installed (psycopg, pymysql, ...).
# SQLite needs nothing beyond the standard library, which is why it is what the
# test suite uses to exercise the connector for real.
sqlalchemy>=2.0

# MongoDB and compatible document stores.
pymongo>=4.6

# S3, MinIO, Cloudflare R2 and other S3-compatible object storage.
boto3>=1.34

# ---------------------------------------------------------------------------
# Local embeddings
# ---------------------------------------------------------------------------
# Only needed if you want embeddings computed in-process rather than by the
# model server you are already running. Wizard prefers the provider's own
# endpoint (Ollama `POST /api/embed`, or `/v1/embeddings` on anything
# OpenAI-compatible) and falls back to a hashing encoder, so this is a
# preference, not a requirement.
#

# not the machine has a GPU. The CPU index is roughly a tenth of that:
#
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
#   pip install sentence-transformers
#
