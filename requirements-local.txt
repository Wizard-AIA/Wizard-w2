# The analysis toolkit, for running WITHOUT Docker.
#
#   uv pip install --system -r requirements.txt -r requirements-local.txt
#
# With `EXECUTION_BACKEND=host`, which is the default, generated code runs in a
# subprocess of *this* interpreter, so it can only
# import what is installed here. Nothing below is needed by the API server, and
# nothing below is needed at all when code runs in a container -- the sandbox
# image carries its own copies.
#
# You do not have to install all of it. The runtime is asked what it can import
# and the model is told exactly that, so a partial install means the agent is
# offered a smaller toolkit -- not that it writes code which then fails.

-i https://pkgs.safetycli.com/repository/nil-2662d/project/ai_data_analyst/pypi/simple/

# SQL over dataframes. Usually the fastest route through a large join.
duckdb==1.5.5

# Multithreaded dataframe execution for large group-bys and joins.
polars==1.43.2

# Statistics, inference and machine learning.
statsmodels==0.14.6
scikit-learn==1.9.0
# The CPU build. Plain `xgboost` ships a CUDA-enabled libxgboost.so: 154 MB
# compressed against 4.5 MB here, for a GPU a local install will not be using.
# xgboost-cpu never publishes macOS wheels (checked back through 3.2.0), which
# means `uv pip install` falls back to a from-source CMake build that fails
# without Homebrew's libomp -- so macOS gets plain `xgboost` instead, which
# ships real macOS wheels and needs no CUDA stripped out on a platform that
# never has CUDA to begin with.
xgboost-cpu==3.4.1 ; sys_platform != 'darwin'
xgboost==3.4.1 ; sys_platform == 'darwin'
lightgbm==4.7.0

# Charting. plotly is what PLOT_FORMAT=html needs; without it the worker is
# told to use matplotlib instead rather than to import something absent.
plotly==7.0.0
seaborn==0.13.2
pillow==12.3.0

# Graphs, and spreadsheet output.
networkx==3.6.1
xlsxwriter==3.2.9
